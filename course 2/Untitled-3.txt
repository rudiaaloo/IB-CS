    lo = 0
    hi = len(array) -1 
    while lo <= hi: 
        mid = (lo + hi) // 2
        val = array[mid]
        if key == val: 
            return True 
        elif key < val: 
            hi = mid - 1 
        else: #key > val 
            low = mid + 1
